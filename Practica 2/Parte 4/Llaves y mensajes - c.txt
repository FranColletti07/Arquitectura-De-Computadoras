org 1000h
mensaje db "ACTIVADO", 10
patron db 81h
org 2000h
mov al, 0FFh
out CA, al
mov al, 0
out CB, al
mov bx, offset mensaje
mov ah, 0; va a almacenar el último leído

        cmp ah, al
