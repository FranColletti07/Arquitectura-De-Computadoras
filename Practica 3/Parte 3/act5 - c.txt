EOI equ 20h 
imr equ 21h
cont equ 10h 
comp equ 11h
vector_int1 equ 10h
int1 equ 25h
org 40h
ptr_handler dw timer_handler
org 1000h
msj db "Vamos las interrupciones!", 10
flag db 0
org 3000H
init:
cli
push ax
mov al, 0FDh
out imr, al
mov al, vector_int1
out int1, al
mov al, 10
out comp, al
mov al, 0
out cont, al
pop ax
sti
ret
timer_handler:
push ax
mov al, 0
out cont, al
mov al, offset flag - offset msj 
mov bx, offset msj
int 7
mov al, 0ffh
out imr, al
mov cl, 1 
mov al, eoi
out eoi, al
pop ax
iret
org 2000H
call init
mov cl, flag
bucle: cmp cl, 1
jnz bucle
int 0
end
